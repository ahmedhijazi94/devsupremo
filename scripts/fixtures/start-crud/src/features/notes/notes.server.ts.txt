import '@tanstack/react-start/server-only'
import { z } from 'zod'
import { requireUser } from '@/features/auth/auth.server'
import { createNoteSchema, filterSchema, idSchema, noteSchema, updateNoteSchema } from './notes.schema'

export async function listOwnNotes(input: z.infer<typeof filterSchema>) {
  const { q } = filterSchema.parse(input)
  const { user, supabase } = await requireUser()
  const query = supabase.from('notes').select('id,title,created_at').eq('user_id', user.id).order('created_at')
  const { data, error } = await (q ? query.ilike('title', `%${q.replaceAll('%', '\\%').replaceAll('_', '\\_')}%`) : query)
  if (error) throw new Error('Não foi possível listar as notas.')
  return z.array(noteSchema).parse(data)
}
export async function readOwnNote(input: z.infer<typeof idSchema>) {
  const { id } = idSchema.parse(input)
  const { user, supabase } = await requireUser()
  const { data, error } = await supabase.from('notes').select('id,title,created_at').eq('id', id).eq('user_id', user.id).maybeSingle()
  if (error || !data) throw new Error('Nota não encontrada.')
  return noteSchema.parse(data)
}
export async function insertOwnNote(input: z.infer<typeof createNoteSchema>) {
  const { title } = createNoteSchema.parse(input)
  const { user, supabase } = await requireUser()
  const { data, error } = await supabase.from('notes').insert({ title, user_id: user.id }).select('id,title,created_at').single()
  if (error) throw new Error('Não foi possível criar a nota.')
  return noteSchema.parse(data)
}
export async function changeOwnNote(input: z.infer<typeof updateNoteSchema>) {
  const { id, title } = updateNoteSchema.parse(input)
  const { user, supabase } = await requireUser()
  const { data, error } = await supabase.from('notes').update({ title }).eq('id', id).eq('user_id', user.id).select('id,title,created_at').maybeSingle()
  if (error || !data) throw new Error('Nota não encontrada.')
  return noteSchema.parse(data)
}
export async function deleteOwnNote(input: z.infer<typeof idSchema>) {
  const { id } = idSchema.parse(input)
  const { user, supabase } = await requireUser()
  const { data, error } = await supabase.from('notes').delete().eq('id', id).eq('user_id', user.id).select('id').maybeSingle()
  if (error || !data) throw new Error('Nota não encontrada.')
  return { deleted: true }
}
