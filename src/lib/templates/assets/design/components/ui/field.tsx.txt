import type { ReactNode } from 'react'
import { cn } from '@/lib/utils'

export function Field({ id, label, hint, error, required, children, className }: {
  id: string; label: string; hint?: string; error?: string; required?: boolean;
  children: ReactNode; className?: string;
}) {
  return <div className={cn('space-y-2', className)}>
    <label htmlFor={id} className="block text-sm font-medium">{label}{required && <span aria-hidden="true" className="text-danger-foreground"> *</span>}</label>
    {children}
    {hint && <p id={`${id}-hint`} className="text-muted text-sm">{hint}</p>}
    {error && <p id={`${id}-error`} role="alert" className="text-danger-foreground text-sm">{error}</p>}
  </div>
}
