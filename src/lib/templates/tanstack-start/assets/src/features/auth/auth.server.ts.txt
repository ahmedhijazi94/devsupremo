import '@tanstack/react-start/server-only'
import { createClient } from '@/lib/supabase/server'
import { getCookies, setResponseHeader, setResponseStatus } from '@tanstack/react-start/server'

export async function requireUser() {
  setResponseHeader('Cache-Control', 'private, no-store')
  if (!Object.keys(getCookies()).some((name) => name.startsWith('sb-'))) { setResponseStatus(401); throw new Error('Autenticação necessária.') }
  const supabase = createClient()
  const { data: { user }, error } = await supabase.auth.getUser()
  if (error || !user) { setResponseStatus(401); throw new Error('Autenticação necessária.') }
  return { user, supabase }
}

export async function currentUser() {
  setResponseHeader('Cache-Control', 'private, no-store')
  if (!Object.keys(getCookies()).some((name) => name.startsWith('sb-'))) return null
  const supabase = createClient()
  const { data: { user }, error } = await supabase.auth.getUser()
  if (error || !user) return null
  return { id: user.id, email: user.email ?? '' }
}
